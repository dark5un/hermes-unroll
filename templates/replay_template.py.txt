def dispatch_tool(name, args, default=None, step=None):
    """Cache lookup with substitute override for tool results."""
    if SUBSTITUTE_TOOL:
        try:
            _s, _j = SUBSTITUTE_TOOL.split(" ", 1)
            if step is not None and int(_s) == step:
                return "[substituted] " + name + "(" + _j + ")"
        except Exception:
            pass
    if step is not None:
        _entry = RESPONSE_CACHE.get("tool_" + str(step))
        if isinstance(_entry, dict) and "result" in _entry:

    return default

def build_langgraph():
    """Build a LangGraph StateGraph from STATE_GRAPH.

    Returns a compiled graph when langgraph is installed, else None.
    Each trace step becomes a node; edges chain consecutive steps.
    """
    if not HAS_LANGGRAPH:
        return None
    builder = StateGraph(dict)
    nodes = STATE_GRAPH.get("nodes", []) if isinstance(STATE_GRAPH, dict) else []
    edges = STATE_GRAPH.get("edges", []) if isinstance(STATE_GRAPH, dict) else []
    for node in nodes:
        nid = str(node.get("id"))

        def _step(state, _nid=nid):
            updated = dict(state)
            updated["last_step"] = _nid
            return updated

        builder.add_node(nid, _step)
    if nodes:
        builder.set_entry_point(str(nodes[0].get("id")))
        for edge in edges:
            builder.add_edge(str(edge.get("from")), str(edge.get("to")))
        builder.add_edge(str(nodes[-1].get("id")), END)
    return builder.compile()
