def dispatch_tool(name, args, default=None, step=None):
    """Cache lookup with substitute override for tool results."""
    if SUBSTITUTE_TOOL:
        try:
            _s, _j = SUBSTITUTE_TOOL.split(" ", 1)
            if step is not None and int(_s) == step:
                return "[substituted] " + name + "(" + _j + ")"
        except Exception:
            pass
    if step is not None:
        _entry = RESPONSE_CACHE.get("tool_" + str(step))
        if isinstance(_entry, dict) and "result" in _entry:
            return _entry["result"]
    return default


def _load_other_timeline(path):
    """Load TIMELINE from another trace file without exec (safe).

    Parses the AST for a top-level TIMELINE assignment and literal-evals
    it, so arbitrary code in the other file never runs.
    """
    import ast as _ast
    with open(path, encoding="utf-8") as _f:
        _src = _f.read()
    try:
        _tree = _ast.parse(_src)
        for _node in _tree.body:
            if isinstance(_node, _ast.Assign):
                for _t in _node.targets:
                    if isinstance(_t, _ast.Name) and _t.id == "TIMELINE":
                        return _ast.literal_eval(_node.value)
    except Exception:
        pass
    raise ValueError("TIMELINE not found in " + str(path))


def _transitive_invalidated(edit_idx):
    """All steps depending transitively on edit_idx via DEPENDENCIES."""
    _norm = {}
    try:
        _items = (DEPENDENCIES or {}).items()
    except AttributeError:
        return [edit_idx]
    for _k, _v in _items:
        try:
            _norm[int(_k)] = [int(_x) for _x in (_v or [])]
        except (TypeError, ValueError):
            continue
    _children = {}
    for _s, _ds in _norm.items():
        for _d in _ds:
            _children.setdefault(_d, []).append(_s)
    invalidated_steps = []
    _queue = [int(edit_idx)]
    _seen = {int(edit_idx)}
    while _queue:
        _cur = _queue.pop(0)
        invalidated_steps.append(_cur)
        for _nxt in _children.get(_cur, []):
            if _nxt not in _seen:
                _seen.add(_nxt)
                _queue.append(_nxt)
    invalidated_steps.sort()
    return invalidated_steps


def _save_edited_trace(result, edit_step, new_text):
    """Write the edited replay to a new file; never overwrites the original.

    The new filename is the original stem plus _edit_<timestamp>. The
    original file is only read, never written.
    """
    _stamp = time.strftime("%Y%m%d_%H%M%S")
    _src = os.path.abspath(__file__)
    _base, _ext = os.path.splitext(_src)
    _dst = _base + "_edit_" + _stamp + (".py" if not _ext else _ext)
    if os.path.abspath(_dst) == _src:
        _dst = _src + ".edit_" + _stamp + ".py"
    with open(_src, encoding="utf-8") as _f:
        _orig = _f.read()
    _header = (
        "\n# EDIT step=" + str(edit_step)
        + " new_text=" + json.dumps(str(new_text)) + "\n"
    )
    with open(_dst, "w", encoding="utf-8") as _f:
        _f.write(_orig + _header)
    print("Saved edited trace: " + _dst)
    return _dst

def build_langgraph():
    """Build a LangGraph StateGraph from STATE_GRAPH.

    Returns a compiled graph when langgraph is installed, else None.
    Each trace step becomes a node; edges chain consecutive steps.
    """
    if not HAS_LANGGRAPH:
        return None
    builder = StateGraph(dict)
    nodes = STATE_GRAPH.get("nodes", []) if isinstance(STATE_GRAPH, dict) else []
    edges = STATE_GRAPH.get("edges", []) if isinstance(STATE_GRAPH, dict) else []
    for node in nodes:
        nid = str(node.get("id"))

        def _step(state, _nid=nid):
            updated = dict(state)
            updated["last_step"] = _nid
            return updated

        builder.add_node(nid, _step)
    if nodes:
        builder.set_entry_point(str(nodes[0].get("id")))
        for edge in edges:
            builder.add_edge(str(edge.get("from")), str(edge.get("to")))
        builder.add_edge(str(nodes[-1].get("id")), END)
    return builder.compile()
